### React Component Example

```typescript
// components/TeamList.tsx
import { useEffect, useState } from 'react';
import type { Team } from '@/types';
import { ENDPOINTS } from '@/types';

export function TeamList() {
  const [teams, setTeams] = useState<Team[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch(`${import.meta.env.VITE_API_URL}${ENDPOINTS.getTeams}`)
      .then(res => res.json())
      .then(data => {
        setTeams(data.teams);
        setLoading(false);
      });
  }, []);

  if (loading) return <div>Loading...</div>;

  return (
    <ul>
      {teams.map(team => (
        <li key={team.id}>{team.name}</li>
      ))}
    </ul>
  );
}
```

### Typed API Client

```typescript
// services/apiClient.ts
import { API_BASE_URL } from '@/types';
import type { ApiResponse, PaginatedResponse } from '@/types';

class ApiClient {
  private baseURL: string;

  constructor() {
    this.baseURL = API_BASE_URL;
  }

  async get<T>(url: string): Promise<ApiResponse<T>> {
    const response = await fetch(`${this.baseURL}${url}`);
    return response.json();
  }

  async post<T>(url: string, data: any): Promise<ApiResponse<T>> {
    const response = await fetch(`${this.baseURL}${url}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    });
    return response.json();
  }

  // Typed methods
  async getTeam(id: string) {
    return this.get<Team>(`/api/v1/teams/${id}`);
  }

  async createTeam(data: TeamCreateRequest) {
    return this.post<Team>('/api/v1/teams', data);
  }
}

export const apiClient = new ApiClient();
```

## 📝 Example: Full Component

```typescript
// components/TeamDetail.tsx
import { useEffect, useState } from 'react';
import { useParams } from 'react-router-dom';
import type {
  Team,
  TeamMember,
  TeamPersonalityMap,
  ApiResponse
} from '@/types';
import { ENDPOINTS } from '@/types';

export function TeamDetail() {
  const { teamId } = useParams<{ teamId: string }>();
  const [team, setTeam] = useState<Team | null>(null);
  const [members, setMembers] = useState<TeamMember[]>([]);
  const [personality, setPersonality] = useState<TeamPersonalityMap | null>(null);

  useEffect(() => {
    async function loadData() {
      // Load team
      const teamRes = await fetch(`${import.meta.env.VITE_API_URL}${ENDPOINTS.getTeam(teamId!)}`);
      const teamData: ApiResponse<Team> = await teamRes.json();
      setTeam(teamData.data);

      // Load members
      const membersRes = await fetch(`${import.meta.env.VITE_API_URL}${ENDPOINTS.getTeam(teamId!)}/members`);
      const membersData: ApiResponse<TeamMember[]> = await membersRes.json();
      setMembers(membersData.data);

      // Load personality data
      const personalityRes = await fetch(`${import.meta.env.VITE_API_URL}${ENDPOINTS.getTeamComposition(teamId!)}`);
      const personalityData: ApiResponse<TeamPersonalityMap> = await personalityRes.json();
      setPersonality(personalityData.data);
    }

    if (teamId) {
      loadData();
    }
  }, [teamId]);

  if (!team) return <div>Loading...</div>;

  return (
    <div>
      <h1>{team.name}</h1>
      <p>{team.description}</p>

      <h2>Team Personality</h2>
      {personality && (
        <div>
          <p>Composition: {personality.composition_type}</p>
          <p>Diversity Score: {personality.diversity_score}</p>
        </div>
      )}

      <h2>Members ({members.length})</h2>
      <ul>
        {members.map(member => (
          <li key={member.id}>{member.role}</li>
        ))}
      </ul>
    </div>
  );
}
```
